function exportContextJSON() {
  const context = getParameters();
  const filename = `predator_prey_${context.seed}_context.json`;
  const blob = new Blob([JSON.stringify(context, null, 2)], { type: "application/json" });
  const link = document.createElement("a");
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  link.click();
}

function importContextJSON() {
  const input = document.createElement('input');
  input.type = 'file';
  input.accept = '.json';
  input.onchange = e => {
    const file = e.target.files[0];
    const reader = new FileReader();
    reader.onload = evt => {
      try {
        const imported = JSON.parse(evt.target.result);
        Object.keys(imported).forEach(key => {
          if (document.getElementById(key)) {
            document.getElementById(key).value = imported[key];
          }
        });
        alert("Context imported.");
      } catch (err) {
        alert("Failed to import context: " + err);
      }
    };
    reader.readAsText(file);
  };
  input.click();
}

  let timer = null, time = 0, data = [], prey, predator, params = {}, paused = false;
  let batchResults = [];

  function updateStatistics() {
    document.getElementById("statistics").innerHTML =
      `<strong>Step:</strong> ${time} &nbsp; | &nbsp; <strong>Prey:</strong> ${prey.toFixed(2)} &nbsp; | &nbsp; <strong>Predator:</strong> ${predator.toFixed(2)}`;
  }

  function togglePauseResume() {
    if (paused) {
      timer = setInterval(stepSimulation, 100);
      document.getElementById("pauseResumeBtn").textContent = "Pause";
    } else {
      clearInterval(timer);
      document.getElementById("pauseResumeBtn").textContent = "Resume";
    }
    paused = !paused;
  }

  function stepSimulation() {
    if (time >= params.runSteps || (prey < params.extinctionThreshold && predator < params.extinctionThreshold)) {
      clearInterval(timer);
      return;
    }
    const dPrey = params.preyBirth * prey * (1 - prey / params.carryingCapacity) - params.predation * prey * predator;
    const dPred = params.conversion * prey * predator - params.predDeath * predator;
    prey += dPrey;
    predator += dPred;
    if (prey < 0) prey = 0;
    if (predator < 0) predator = 0;
    data.push({ time, prey, predator });
    time++;
    updateChart();
    updateStatistics();
  }

  function continueSimulation() {
    let totalSteps = parseInt(document.getElementById("totalSteps").value)
    params.runSteps = params.runSteps + totalSteps;
    if (timer) clearInterval(timer);
    timer = setInterval(stepSimulation, 100);
  }

  function updateChart() {
    const times = data.map(d => d.time);
    const preyPop = data.map(d => d.prey);
    const predPop = data.map(d => d.predator);

    const trace1 = {
      x: times,
      y: preyPop,
      mode: 'lines',
      name: 'Prey',
      line: { color: 'green' }
    };
    const trace2 = {
      x: times,
      y: predPop,
      mode: 'lines',
      name: 'Predator',
      line: { color: 'red' }
    };
    const layout = {
      title: 'Predator–Prey Population Over Time',
      xaxis: { title: 'Time Step' },
      yaxis: { title: 'Population' }
    };
    Plotly.newPlot('chart', [trace1, trace2], layout);
  }

  function startSimulation() {
    params = getParameters();
    params.runSteps = params.totalSteps;
    data = [];
    prey = params.initialPrey;
    predator = params.initialPred;
    time = 0;
    paused = false;
    document.getElementById("pauseResumeBtn").textContent = "Pause";

    if (timer) clearInterval(timer);
    timer = setInterval(stepSimulation, 100);
    updateChart();
    updateStatistics();
  }

  function resetSimulation() {
    if (timer) clearInterval(timer);
    data = [];
    prey = predator = time = 0;
    updateChart();
    document.getElementById("pauseResumeBtn").textContent = "Pause";
    paused = false;
    updateStatistics();
  }

  function getParameters() {
    return {
      preyBirth: parseFloat(document.getElementById("preyBirth").value),
      predDeath: parseFloat(document.getElementById("predDeath").value),
      predation: parseFloat(document.getElementById("predation").value),
      conversion: parseFloat(document.getElementById("conversion").value),
      initialPrey: parseFloat(document.getElementById("initialPrey").value),
      initialPred: parseFloat(document.getElementById("initialPred").value),
      carryingCapacity: parseFloat(document.getElementById("carryingCapacity").value),
      extinctionThreshold: parseFloat(document.getElementById("extinctionThreshold").value),
      seed: document.getElementById("seed").value,
      totalSteps: parseInt(document.getElementById("totalSteps").value)
    };
  }

  function exportResultsCSV() {
    const filename = `predator_prey_${params.seed}_results.csv`;
    const csvContent = "data:text/csv;charset=utf-8," +
      ["time,prey,predator"].concat(data.map(d => `${d.time},${d.prey},${d.predator}`)).join("\n");
    const encodedUri = encodeURI(csvContent);
    const link = document.createElement("a");
    link.setAttribute("href", encodedUri);
    link.setAttribute("download", filename);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  }

  function downloadChart(format) {
    const filename = `predator_prey_${params.seed}_chart.${format}`;
    Plotly.downloadImage(document.getElementById('chart'), {
      format: format,
      filename: filename,
      height: 400,
      width: 800
    });
  }

  function runBatchSimulations() {
    const runs = parseInt(document.getElementById("batchRuns").value);
    const baseParams = getParameters();
    batchResults = [];
    for (let i = 0; i < runs; i++) {
      let result = simulateBatchRun({ ...baseParams }, i);
      batchResults.push(result);
    }
    alert(`Batch completed: ${runs} simulations.`);
  }

  function simulateBatchRun(runParams, index) {
    let batchData = [];
    let prey = runParams.initialPrey;
    let predator = runParams.initialPred;
    for (let t = 0; t < runParams.totalSteps; t++) {
      const dPrey = runParams.preyBirth * prey * (1 - prey / runParams.carryingCapacity) - runParams.predation * prey * predator;
      const dPred = runParams.conversion * prey * predator - runParams.predDeath * predator;
      prey += dPrey;
      predator += dPred;
      if (prey < 0) prey = 0;
      if (predator < 0) predator = 0;
      batchData.push({ time: t, prey, predator });
      if (prey < runParams.extinctionThreshold && predator < runParams.extinctionThreshold) break;
    }
    return {
      run: index,
      seed: runParams.seed + "_run" + index,
      parameters: runParams,
      results: batchData
    };
  }

  function downloadBatchResults() {
    const filename = `predator_prey_${params.seed}_batch.json`;
    const blob = new Blob([JSON.stringify(batchResults, null, 2)], { type: "application/json" });
    const link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.download = filename;
    link.click();
  }

  function downloadBatchCSV() {
    let rows = ["run,time,prey,predator"];
    batchResults.forEach(sim => {
      sim.results.forEach(row => {
	rows.push(`${sim.run},${row.time},${row.prey},${row.predator}`);
      });
    });
    const filename = `predator_prey_${params.seed}_batch.csv`;
    const blob = new Blob([rows.join("\n")], { type: "text/csv" });
    const link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.download = filename;
    link.click();
  }

  function getParameters() {
    return {
      preyBirth: parseFloat(document.getElementById("preyBirth").value),
      predDeath: parseFloat(document.getElementById("predDeath").value),
      predation: parseFloat(document.getElementById("predation").value),
      conversion: parseFloat(document.getElementById("conversion").value),
      initialPrey: parseFloat(document.getElementById("initialPrey").value),
      initialPred: parseFloat(document.getElementById("initialPred").value),
      carryingCapacity: parseFloat(document.getElementById("carryingCapacity").value),
      extinctionThreshold: parseFloat(document.getElementById("extinctionThreshold").value),
      seed: document.getElementById("seed").value
    };
  }

  let data = [];
  let time = 0;
  let isPaused = false;
  let chartLayout = {
    title: 'Predator–Prey Dynamics',
    xaxis: { title: 'Time Steps' },
    yaxis: { title: 'Population' },
  };

  function startSimulation() {
    data = [];
    time = 0;
    isPaused = false;
    const params = getParameters();
    data.push({ time: 0, prey: params.initialPrey, pred: params.initialPred });
    updateChart();
    requestAnimationFrame(simulationStep);
  }

  function simulationStep() {
    if (isPaused) return;
    const params = getParameters();
    const last = data[data.length - 1];
    let prey = last.prey;
    let pred = last.pred;

    prey += params.preyBirth * prey * (1 - prey / params.carryingCapacity) - params.predation * prey * pred;
    pred += params.conversion * params.predation * prey * pred - params.predDeath * pred;

    prey = Math.max(0, prey);
    pred = Math.max(0, pred);

    time++;
    data.push({ time, prey, pred });
    updateChart();
    updateStatistics();

    if (prey > params.extinctionThreshold || pred > params.extinctionThreshold) {
      requestAnimationFrame(simulationStep);
    }
  }

  function togglePauseResume() {
    isPaused = !isPaused;
    if (!isPaused) requestAnimationFrame(simulationStep);
    document.getElementById("pauseResumeBtn").innerText = isPaused ? "Resume" : "Pause";
  }

  function continueSimulation() {
    isPaused = false;
    requestAnimationFrame(simulationStep);
  }

  function resetSimulation() {
    data = [];
    time = 0;
    Plotly.newPlot("chart", [], chartLayout);
    document.getElementById("statistics").innerText = "";
  }

  function updateChart() {
    const times = data.map(d => d.time);
    const prey = data.map(d => d.prey);
    const pred = data.map(d => d.pred);

    const tracePrey = { x: times, y: prey, name: 'Prey', type: 'scatter' };
    const tracePred = { x: times, y: pred, name: 'Predator', type: 'scatter' };

    Plotly.newPlot("chart", [tracePrey, tracePred], chartLayout);
  }

  function updateStatistics() {
    const last = data[data.length - 1];
    document.getElementById("statistics").innerText =
      `Step: ${last.time}, Prey: ${last.prey.toFixed(2)}, Predator: ${last.pred.toFixed(2)}`;
  }

  function downloadChart(format) {
    const params = getParameters();
    const filename = `predator_prey_${params.seed}_chart.${format}`;
    Plotly.downloadImage(document.getElementById('chart'), {
      format,
      filename,
      height: 400,
      width: 800
    });
  }

  function exportResultsCSV() {
    const params = getParameters();
    const filename = `predator_prey_${params.seed}_results.csv`;
    const header = "time,prey,predator\n";
    const body = data.map(d => `${d.time},${d.prey.toFixed(4)},${d.pred.toFixed(4)}`).join("\n");
    const blob = new Blob([header + body], { type: "text/csv" });
    const link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.download = filename;
    link.click();
  }
